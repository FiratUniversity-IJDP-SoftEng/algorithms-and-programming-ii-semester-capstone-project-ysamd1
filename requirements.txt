import math

def clean_step_data(step):
    """
    Her adım verisini temizler.

    - Eğer bir uzaklık değeri sonsuzsa (math.inf), bunu 'inf' olarak değiştirir.
    - Eğer bir alan None (boş) ise, bunu boş string ('') yapar.

    Bu işlemler, adımların görsel sunum veya JSON çıktısında düzgün görünmesini sağlar.
    """
    for key in step.get("dist", {}).keys():
        if step["dist"][key] == math.inf:
            step["dist"][key] = 'inf'

    for key in step.keys():
        if step[key] is None:
            step[key] = ''

    return step


def convert_queue_format(step):
    """
    Kuyruk içeriğini (öncelik, düğüm) formatından (düğüm, öncelik) formatına dönüştürür.

    Görselleştirme sırasında daha anlaşılır olması için kullanılır.
    Örneğin: (2, 'B') → ['B', 2]
    """
    new_queue = []
    for node in step.get("queue", []):
        new_queue.append([node[1], node[0]])
    step["queue"] = new_queue
    return step
